## 1. Контроллер Камеры (`node_camera_controller.gd`)

* **Назначение:** Единый контроллер `Camera3D`, управляющий перемещением между Узлами (Nodes), пассивным слежением за мышью (`Head Look`) с мёртвой зоной (`Deadzone`) и сглаживанием, а также суб-фокусировкой.



```gdscript
# res://scripts/camera/node_camera_controller.gd
class_name NodeCameraController
extends Camera3D

enum State { IDLE_LOOK, TRANSITION, SUB_NODE_FOCUS, INSPECT_ITEM }

var current_state: State = State.IDLE_LOOK

@export_group("Head Look Settings")
@export var max_yaw_deg: float = 10.0
@export var max_pitch_deg: float = 6.0
@export var deadzone_size: Vector2 = Vector2(0.25, 0.25) # 25% в центре экрана
@export var look_smooth_speed: float = 5.0

@export_group("Transition Settings")
@export var transition_duration: float = 0.4

var base_rotation: Vector3
var target_rotation: Vector3

func _process(delta: float) -> void:
	match current_state:
		State.IDLE_LOOK:
			_update_head_look(delta)

func _update_head_look(delta: float) -> void:
	var viewport_size = get_viewport().get_visible_rect().size
	var mouse_pos = get_viewport().get_mouse_position()
	
	# Нормализация координат от -1 до 1
	var norm_mouse = (mouse_pos / viewport_size) * 2.0 - Vector2.ONE
	
	# Проверка Deadzone
	if abs(norm_mouse.x) < deadzone_size.x:
		norm_mouse.x = 0.0
	else:
		norm_mouse.x = sign(norm_mouse.x) * ((abs(norm_mouse.x) - deadzone_size.x) / (1.0 - deadzone_size.x))
		
	if abs(norm_mouse.y) < deadzone_size.y:
		norm_mouse.y = 0.0
	else:
		norm_mouse.y = sign(norm_mouse.y) * ((abs(norm_mouse.y) - deadzone_size.y) / (1.0 - deadzone_size.y))
		
	# Расчёт целевого поворота
	var yaw = -norm_mouse.x * deg_to_rad(max_yaw_deg)
	var pitch = -norm_mouse.y * deg_to_rad(max_pitch_deg)
	
	var offset_rot = Vector3(pitch, yaw, 0.0)
	
	rotation.x = lerp_angle(rotation.x, base_rotation.x + offset_rot.x, look_smooth_speed * delta)
	rotation.y = lerp_angle(rotation.y, base_rotation.y + offset_rot.y, look_smooth_speed * delta)

func move_to_node(target_transform: Transform3D, duration: float = transition_duration) -> void:
	current_state = State.TRANSITION
	
	var tween = create_tween().set_parallel(true).set_trans(Tween.TRANS_QUAD).set_ease(Tween.EASE_IN_OUT)
	tween.tween_property(self, "global_transform", target_transform, duration)
	
	tween.finished.connect(func():
		base_rotation = rotation
		current_state = State.IDLE_LOOK
	)

```

## 2. Компонент Интерактивности (`hover_interactable.gd`)

* **Назначение:** Прикрепляемый узел-компонент (`Attachable Component`), вешаемый на любой интерактивный 3D/2D объект в сцене для обработки наведения, тайминга `Dwell Time` и вызова аудио/визуального фидбека.



```gdscript
# res://scripts/interaction/hover_interactable.gd
class_name HoverInteractable
extends Area3D

signal hover_triggered
signal hover_canceled

@export_group("Hover Timings")
@export var dwell_time: float = 0.15 # Задержка срабатывания (150 мс)
@export var is_swat_target: bool = false # Флаг для особых анимаций (например, Управдом)

@export_group("Feedback Effects")
@export var hover_sound: AudioStream
@export var highlight_material: ShaderMaterial

var _hover_timer: SceneTreeTimer
var _is_hovered: bool = false

func _ready() -> void:
	mouse_entered.connect(_on_mouse_entered)
	mouse_exited.connect(_on_mouse_exited)

func _on_mouse_entered() -> void:
	_is_hovered = true
	# Запуск таймера проверки Dwell Time
	get_tree().create_timer(dwell_time).timeout.connect(func():
		if _is_hovered:
			_activate_hover_feedback()
	)

func _on_mouse_exited() -> void:
	_is_hovered = false
	hover_canceled.emit()
	_deactivate_hover_feedback()

func _activate_hover_feedback() -> void:
	hover_triggered.emit()
	if hover_sound:
		AudioManager.play_hover_sfx(hover_sound) # Проигрывание через шину с кулдауном
	if is_swat_target:
		# Триггер логики отмахивания руки Управдома
		get_parent().trigger_swat_animation()

```

## 3. Логика Анимации Руки Управдома (`landlord_hand.gd`)

* **Назначение:** Обработка реакций и анимаций руки [[Управдом]]а при попытке игрока отмахнуться от него или затянуть передачу денег.



```gdscript
# res://scripts/entities/landlord_hand.gd
extends Node3D

var is_waiting_for_money: bool = true
var is_swatting: bool = false
@onready var anim_player: AnimationPlayer = $AnimationPlayer

func trigger_swat_animation() -> void:
	if is_waiting_for_money and not is_swatting:
		is_swatting = true
		anim_player.play("swat_fly_annoyed")
		await anim_player.animation_finished
		is_swatting = false
		anim_player.play("idle_open_palm")

```
